def sum_of_faces():
    # Define the variables
    a, b, c, d, e, f = symbols('a b c d e f')
    
    # Define the sum of the numbers on the vertices
    sum_vertices = a*b*c + a*b*d + a*b*e + a*b*f + \
                    a*c*e + a*c*f + b*c*e + b*c*f + \
                    b*d*e + b*d*f + c*d*e + c*d*f + \
                    d*e*f
    
    # Set the sum of the numbers on the vertices equal to 1001
    equation = Eq(sum_vertices, 1001)
    
    # Solve the equation
    solution = solve(equation, (a, b, c, d, e, f))
    
    # Calculate the sum of the numbers written on the faces
    sum_faces = a + b + c + d + e + f
    
    # Return the sum of the numbers written on the faces
    return sum_faces.subs(solution[0])
